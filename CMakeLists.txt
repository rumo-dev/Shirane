cmake_minimum_required(VERSION 3.25)
project(Shirane VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Shirane requires Windows x64 and MSVC v143")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
function(shirane_target target)
  target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_include_directories(${target} PUBLIC "${PROJECT_SOURCE_DIR}/src")
endfunction()
find_package(nlohmann_json CONFIG REQUIRED)
find_package(spdlog CONFIG REQUIRED)
find_package(GTest CONFIG REQUIRED)
include(FetchContent)
FetchContent_Declare(imgui
  GIT_REPOSITORY https://github.com/ocornut/imgui.git
  GIT_TAG 52fe0a05a7b1aa180a202bb24f0f2a049a9c1b7d)
FetchContent_MakeAvailable(imgui)
add_library(shirane_imgui STATIC
  ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
  ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_dx11.cpp)
target_include_directories(shirane_imgui SYSTEM PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_link_libraries(shirane_imgui PRIVATE d3d11 dxgi d3dcompiler dwmapi)
add_library(sas_audio_core STATIC src/audio/core/AudioCore.cpp)
shirane_target(sas_audio_core)
add_library(sas_config STATIC src/config/Profile.cpp)
shirane_target(sas_config)
target_link_libraries(sas_config PRIVATE nlohmann_json::nlohmann_json)
add_library(sas_ui STATIC src/ui/MixerView.cpp)
shirane_target(sas_ui)
target_link_libraries(sas_ui PUBLIC sas_config shirane_imgui)
add_library(sas_platform STATIC src/platform/DeviceService.cpp)
shirane_target(sas_platform)
target_link_libraries(sas_platform PRIVATE ole32 uuid)
add_executable(shirane_device_probe tests/DeviceProbe.cpp)
shirane_target(shirane_device_probe)
target_link_libraries(shirane_device_probe PRIVATE sas_platform)
option(SHIRANE_ENABLE_HARDWARE_TESTS "Register read-only Windows audio device tests" OFF)
add_executable(Shirane WIN32 src/app/Main.cpp src/app/Shirane.manifest)
shirane_target(Shirane)
target_link_libraries(Shirane PRIVATE sas_ui sas_audio_core sas_platform spdlog::spdlog d3d11 dxgi shell32 ole32)
target_link_options(Shirane PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${PROJECT_SOURCE_DIR}/src/app/Shirane.manifest")
enable_testing()
if(SHIRANE_ENABLE_HARDWARE_TESTS)
  add_test(NAME DeviceService.RealMachine COMMAND shirane_device_probe)
  set_tests_properties(DeviceService.RealMachine PROPERTIES LABELS "hardware" SKIP_RETURN_CODE 77 TIMEOUT 30)
endif()
add_executable(sas_tests tests/ProfileTests.cpp tests/DeviceCheckTests.cpp)
shirane_target(sas_tests)
target_link_libraries(sas_tests PRIVATE sas_config sas_audio_core GTest::gtest_main nlohmann_json::nlohmann_json)
include(GoogleTest)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/test-data")
gtest_discover_tests(sas_tests PROPERTIES LABELS unit
  ENVIRONMENT "SHIRANE_TEST_ROOT=${CMAKE_BINARY_DIR}/test-data")
